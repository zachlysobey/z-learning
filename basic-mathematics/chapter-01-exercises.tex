\section{Rules for Addition}

Justify each step, using commutativity and associativity in proving the
following identities.

\begin{enumerate}
  \item $(a + b) + (c + d) = (a + d) + (b + c)$

    \emph{Solution.}
    \begin{align*}
      (a + b) + (c + d)
        &= a + (b + (c + d)) && \text{associativity} \\
        &= a + ((b + c) + d) && \text{associativity} \\
        &= a + (d + (b + c)) && \text{commutativity} \\
        &= (a + d) + (b + c) && \text{associativity}
    \end{align*}

  \item $(a + b) + (c + d) = (a + c) + (b + d)$

    \emph{Solution.}
    \begin{align*}
      (a + b) + (c + d)
        &= a + (b + (c + d)) && \text{associativity} \\
        &= a + ((c + d) + b) && \text{commutativity} \\
        &= a + (c + (d + b)) && \text{associativity} \\
        &= (a + c) + (d + b) && \text{associativity} \\
        &= (a + c) + (b + d) && \text{commutativity}
    \end{align*}

  \item $(a - b) + (c - d) = (a + c) + (-b - d)$

    \emph{Solution.}
    \begin{align*}
      (a - b) + (c - d)
        &= a + (-b + (c - d)) && \text{associativity} \\
        &= a + (-b + (-d + c)) && \text{commutativity} \\
        &= a + ((-b - d) + c) && \text{associativity} \\
        &= a + (c + (-b - d)) && \text{commutativity} \\
        &= (a + c) + (-b - d) && \text{associativity}
    \end{align*}

  \item $(a - b) + (c - d) = (a + c) - (b + d)$

    \emph{Solution.}
    \begin{align*}
      (a - b) + (c - d)
        &= a + (-b + (c - d)) && \text{associativity} \\
        &= a + ((c - d) - b) && \text{commutativity} \\
        &= a + (c + (-d - b)) && \text{associativity} \\
        &= (a + c) + (-d - b) && \text{associativity} \\
        &= (a + c) + (-b - d) && \text{commutativity} \\
        &= (a + c) - (b + d) && \text{N5}
    \end{align*}

  \item $(a - b) + (c - d) = (a - d) + (c - b)$

    \emph{Solution.}
    \begin{align*}
      (a - b) + (c - d)
        &= a + (-b + (c - d)) && \text{associativity} \\
        &= a + ((c - d) - b) && \text{commutativity} \\
        &= a + ((-d + c) - b) && \text{commutativity} \\
        &= a + (-d + (c - b)) && \text{associativity} \\
        &= (a - d) + (c - b) && \text{associativity}
    \end{align*}

  \item $(a - b) + (c - d) = -(b + d) + (a + c)$

    \emph{Solution.}
    \begin{align*}
      (a - b) + (c - d)
        &= (-b + a) + (c - d) && \text{commutativity} \\
        &= -b + (a + (c - d)) && \text{associativity} \\
        &= -b + ((a + c) - d) && \text{associativity} \\
        &= -b + (-d + (a + c)) && \text{commutativity} \\
        &= (-b - d) + (a + c) && \text{associativity} \\
        &= -(b + d) + (a + c) && \text{N5}
    \end{align*}

  \item $(a - b) + (c - d) = -(b + d) - (-a - c)$

    \emph{Solution.}
    \begin{align*}
      (a - b) + (c - d)
        &= (-b + a) + (c - d) && \text{commutativity} \\
        &= -b + (a + (c - d)) && \text{associativity} \\
        &= -b + ((a + c) - d) && \text{associativity} \\
        &= -b + (-d + (a + c)) && \text{commutativity} \\
        &= (-b - d) + (a + c) && \text{associativity} \\
        &= -(b + d) + (a + c) && \text{N5} \\
        &= -(b + d) + (-(-a) - (-c)) && \text{N4} \\
        &= -(b + d) - (-a - c) && \text{N5}
    \end{align*}

  \item $((x + y) + z) + w = (x + z) + (y + w)$

    \emph{Solution.}
    \begin{align*}
      ((x + y) + z) + w
        &= (x + (y + z)) + w && \text{associativity} \\
        &= (x + (z + y)) + w && \text{commutativity} \\
        &= ((x + z) + y) + w && \text{associativity} \\
        &= (x + z) + (y + w) && \text{associativity}
    \end{align*}

  \item $(x - y) - (z - w) = (x + w) - y - z$

    \emph{Solution.}
    \begin{align*}
      (x - y) - (z - w)
        &= x + (-y - (z - w)) && \text{associativity} \\
        &= x + (-y + (-z - (-w))) && \text{N5} \\
        &= x + ((-y - z) - (-w)) && \text{associativity} \\
        &= x + ((-y - z) + w) && \text{N4} \\
        &= x + (w + (-y - z)) && \text{commutativity} \\
        &= (x + w) + (-y - z) && \text{associativity} \\
        &= (x + w) - y - z && \text{associativity}
    \end{align*}

  \item $(x - y) - (z - w) = (x - z) + (w - y)$
  \item Show that $-(a + b + c) = -a + (-b) + (-c)$.
  \item Show that $-(a - b - c) = -a + b + c$.
  \item Show that $-(a - b) = b - a$.
\end{enumerate}

Solve for $x$ in the following equations.

\begin{enumerate}[resume]
  \item $-2 + x = 4$
  \item $2 - x = 5$
  \item $x - 3 = 7$
  \item $-x + 4 = -1$
  \item $4 - x = 8$
  \item $-5 - x = -2$
  \item $-7 + x = -10$
  \item $-3 + x = 4$
\end{enumerate}

\begin{enumerate}[resume]
  \item Prove the cancellation law for addition:
    if $a + b = a + c$, then $b = c$.
  \item Prove: if $a + b = a$, then $b = 0$.
\end{enumerate}
